  <script>
    const GOOGLE_CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vSIZgkpubsKd4olfELOU2Gglupj1OsgnuaowjKcQZE2vs9ZFqKw7IlivJr0c_jNJsPWbvuOfwnQLjr6/pub?gid=0&single=true&output=csv";

    fetch(GOOGLE_CSV_URL)
      .then(res => res.text())
      .then(csvText => {
        const rows = csvText.split('\n').map(row => row.split(','));
        const tbody = document.querySelector("#admin-table tbody");
        tbody.innerHTML = "";

        for(let i = 1; i < rows.length; i++) {
          if(rows[i].length < 5) continue;
          const tr = document.createElement("tr");
          tr.innerHTML = `
            <td>${rows[i][0] || ''}</td>
            <td>${rows[i][1] || ''}</td>
            <td>${rows[i][2] || ''}</td>
            <td>${rows[i][3] || ''}</td>
            <td>${rows[i][4] || ''}</td>
            <td><b>${rows[i][5] || ''}</b></td>
            <td>${rows[i][7] || ''}</td>
          `;
          tbody.appendChild(tr);
        }
      });
  </script>
